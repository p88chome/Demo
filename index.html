import urllib.request import urllib.parse import urllib.error import json TOKEN_URL = "http://f12aqiids.umc.com/connect/token" API_URL = "http://f12aqigw.umc.com/PI-Query/api/DataQueryAPI/GetQueryDataByJson" SQLID = "DQ-GFYPT-20260908-083610218884-MPZMJ" CLIENT_ID = "ADTUSER" CLIENT_SECRET = "ADTPROJECT01" SCOPE = "" # 如果知道 scope 就填在這裡，不知道留空即可 def get_token(): payload = { "grant_type": "client_credentials", "client_id": CLIENT_ID, "client_secret": CLIENT_SECRET, } if SCOPE: payload["scope"] = SCOPE data = urllib.parse.urlencode(payload).encode("utf-8") req = urllib.request.Request( TOKEN_URL, data=data, method="POST", headers={"Content-Type": "application/x-www-form-urlencoded"} ) with urllib.request.urlopen(req, timeout=30) as resp: result = json.loads(resp.read().decode("utf-8")) return result["access_token"] def try_call(label, headers): print("=" * 60) print(f"[{label}]") print("Headers:", {k: (v[:25] + "..." if k == "Authorization" else v) for k, v in headers.items()}) url = f"{API_URL}?strSQLID={SQLID}" req = urllib.request.Request(url, headers=headers) try: with urllib.request.urlopen(req, timeout=30) as resp: print("狀態:", resp.status, "→ 成功!") data = json.loads(resp.read().decode("utf-8")) print(json.dumps(data, ensure_ascii=False, indent=2)[:1500]) return True except urllib.error.HTTPError as e: print("失敗:", e.code, e.reason) body = e.read().decode("utf-8", errors="replace") if body: print("Body:", body) return False print("正在取得 token...") token = get_token() print("Token 取得成功\n") base_auth = f"Bearer {token}" tests = [ ("1. 目前用法 (Authorization + Accept)", { "Authorization": base_auth, "Accept": "application/json", }), ("2. 加上 User-Agent 模擬瀏覽器", { "Authorization": base_auth, "Accept": "application/json", "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64)", }), ("3. Accept 改成 */* (跟 curl 預設一致)", { "Authorization": base_auth, "Accept": "*/*", }), ("4. 拿掉 Accept，只留 Authorization", { "Authorization": base_auth, }), ("5. 加上 Accept-Encoding + Connection (模擬完整瀏覽器行為)", { "Authorization": base_auth, "Accept": "*/*", "Accept-Encoding": "gzip, deflate", "Connection": "keep-alive", "User-Agent": "PostmanRuntime/7.36.0", }), ("6. 加上 Host 明確指定 (通常自動帶，這裡手動測試)", { "Authorization": base_auth, "Accept": "application/json", "Host": "f12aqigw.umc.com", }), ("7. Content-Type 也一併帶上 (即使GET通常不需要)", { "Authorization": base_auth, "Accept": "application/json", "Content-Type": "application/json", }), ("8. 完全模擬 Postman 預設 headers 組合", { "Authorization": base_auth, "Accept": "*/*", "Accept-Encoding": "gzip, deflate, br", "User-Agent": "PostmanRuntime/7.36.0", "Cache-Control": "no-cache", "Connection": "keep-alive", }), ] success_found = False for label, headers in tests: ok = try_call(label, headers) if ok: success_found = True print("\n*** 這個組合成功了，記下這個 headers 組合! ***\n") if not success_found: print("=" * 60) print("所有組合都失敗，問題應該不在 header 細節上，") print("建議明天請對方提供 Postman 的完整 curl code 逐字比對。")
