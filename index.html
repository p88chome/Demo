import urllib.request import urllib.parse import urllib.error import json TOKEN_URL = "http://f12aqiids.umc.com/connect/token" API_URL = "http://f12aqigw.umc.com/PI-Query/api/DataQueryAPI/GetQueryDataByJson" SQLID = "DQ-GFYPT-20260908-083610218884-MPZMJ" USER = "ADTUSER" PASS = "ADTPROJECT01" def try_token(label, payload): print("=" * 60) print(f"[{label}]") print("Payload:", {k: (v if k not in ("password", "client_secret") else "****") for k, v in payload.items()}) data = urllib.parse.urlencode(payload).encode("utf-8") req = urllib.request.Request( TOKEN_URL, data=data, method="POST", headers={"Content-Type": "application/x-www-form-urlencoded"} ) try: with urllib.request.urlopen(req, timeout=30) as resp: result = json.loads(resp.read().decode("utf-8")) print("成功! 拿到 token") print("expires_in:", result.get("expires_in")) print("scope:", result.get("scope")) return result.get("access_token") except urllib.error.HTTPError as e: print("失败:", e.code, e.reason) body = e.read().decode("utf-8", errors="replace") print("Body:", body) return None def try_query(token, label): print("-" * 60) print(f"用這個 token 查詢資料 [{label}]") url = f"{API_URL}?strSQLID={SQLID}" req = urllib.request.Request( url, headers={"Authorization": f"Bearer {token}", "Accept": "application/json"} ) try: with urllib.request.urlopen(req, timeout=30) as resp: print("status:", resp.status) data = json.loads(resp.read().decode("utf-8")) print(json.dumps(data, ensure_ascii=False, indent=2)[:1500]) return True except urllib.error.HTTPError as e: print("查詢失敗:", e.code, e.reason) print("Body:", e.read().decode("utf-8", errors="replace")) return False # 測試A：password grant，client_id 跟 username 用同一個值，沒有 client_secret token_a = try_token("測試A: password grant, client_id=username, 無 client_secret", { "grant_type": "password", "username": USER, "password": PASS, "client_id": USER, }) if token_a: try_query(token_a, "測試A") # 測試B：password grant，client_id 跟 username 用同一個值，client_secret 用 password 值 token_b = try_token("測試B: password grant, client_id=username, client_secret=password", { "grant_type": "password", "username": USER, "password": PASS, "client_id": USER, "client_secret": PASS, }) if token_b: try_query(token_b, "測試B") # 測試C：password grant，只給 username/password，不給 client_id（有些寬鬆設定允許這樣） token_c = try_token("測試C: password grant, 只有 username/password", { "grant_type": "password", "username": USER, "password": PASS, }) if token_c: try_query(token_c, "測試C")
