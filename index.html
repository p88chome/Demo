WITH d AS ( SELECT MBLNR, MJAHR FROM dbo.MKPF GROUP BY MBLNR, MJAHR HAVING COUNT(*) > 1 ) SELECT (SELECT COUNT(*) FROM d) AS 重複的鍵數, (SELECT COUNT(*) FROM (SELECT DISTINCT h.* FROM dbo.MKPF h JOIN d ON d.MBLNR = h.MBLNR AND d.MJAHR = h.MJAHR) x ) AS 不同的列數; SELECT MJAHR, LEFT(CONVERT(varchar(8), BUDAT, 112), 6) AS 過帳年月, COUNT(*) AS 重複的鍵數 FROM (SELECT MBLNR, MJAHR, MAX(BUDAT) AS BUDAT FROM dbo.MKPF GROUP BY MBLNR, MJAHR HAVING COUNT(*) > 1) d GROUP BY MJAHR, LEFT(CONVERT(varchar(8), BUDAT, 112), 6) ORDER BY MJAHR, 過帳年月; -- 備份多出來的那一份 WITH r AS ( SELECT *, ROW_NUMBER() OVER (PARTITION BY MBLNR, MJAHR ORDER BY (SELECT NULL)) AS rn FROM dbo.MKPF ) SELECT * INTO dbo.MKPF_deleted_dup FROM r WHERE rn > 1; -- 每組留一筆 BEGIN TRAN; WITH r AS ( SELECT ROW_NUMBER() OVER (PARTITION BY MBLNR, MJAHR ORDER BY (SELECT NULL)) AS rn FROM dbo.MKPF ) DELETE FROM r WHERE rn > 1; -- 刪除筆數要等於備份筆數，對了就 COMMIT，不對就 ROLLBACK SELECT @@ROWCOUNT AS 刪除筆數, (SELECT COUNT(*) FROM dbo.MKPF_deleted_dup) AS 備份筆數; -- COMMIT; / ROLLBACK; EXEC etl.Load_MaterialMovement;
