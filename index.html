SELECT 'MSEG' AS 表, COUNT(*) AS 重複的鍵數, SUM(c - 1) AS 多出來的列數 FROM (SELECT COUNT(*) c FROM dbo.MSEG GROUP BY MBLNR, MJAHR, ZEILE HAVING COUNT(*) > 1) d UNION ALL SELECT 'MKPF', COUNT(*), SUM(c - 1) FROM (SELECT COUNT(*) c FROM dbo.MKPF GROUP BY MBLNR, MJAHR HAVING COUNT(*) > 1) d UNION ALL SELECT 'MARA', COUNT(*), SUM(c - 1) FROM (SELECT COUNT(*) c FROM dbo.MARA GROUP BY MATNR HAVING COUNT(*) > 1) d UNION ALL SELECT 'EKKO', COUNT(*), SUM(c - 1) FROM (SELECT COUNT(*) c FROM dbo.EKKO GROUP BY EBELN HAVING COUNT(*) > 1) d UNION ALL SELECT 'LFA1', COUNT(*), SUM(c - 1) FROM (SELECT COUNT(*) c FROM dbo.LFA1 GROUP BY LIFNR HAVING COUNT(*) > 1) d; WITH d AS ( SELECT MBLNR, MJAHR, ZEILE FROM dbo.MSEG GROUP BY MBLNR, MJAHR, ZEILE HAVING COUNT(*) > 1 ) SELECT (SELECT COUNT(*) FROM d) AS 重複的鍵數, (SELECT COUNT(*) FROM (SELECT DISTINCT s.* FROM dbo.MSEG s JOIN d ON d.MBLNR = s.MBLNR AND d.MJAHR = s.MJAHR AND d.ZEILE = s.ZEILE) x ) AS 不同的列數; -- 備份多出來的那一份 WITH r AS ( SELECT *, ROW_NUMBER() OVER (PARTITION BY MBLNR, MJAHR, ZEILE ORDER BY (SELECT NULL)) AS rn FROM dbo.MSEG ) SELECT * INTO dbo.MSEG_deleted_dup FROM r WHERE rn > 1; -- 每組留一筆 BEGIN TRAN; WITH r AS ( SELECT ROW_NUMBER() OVER (PARTITION BY MBLNR, MJAHR, ZEILE ORDER BY (SELECT NULL)) AS rn FROM dbo.MSEG ) DELETE FROM r WHERE rn > 1; -- 刪掉的筆數要等於 MSEG_deleted_dup 的筆數，對了就 COMMIT，不對就 ROLLBACK SELECT @@ROWCOUNT AS 刪除筆數, (SELECT COUNT(*) FROM dbo.MSEG_deleted_dup) AS 備份筆數; -- COMMIT; / ROLLBACK; EXEC etl.Load_MaterialMovement; EXEC etl.Load_MaterialMovement;
