import base64, json, urllib.request, urllib.error def decode_jwt(tok): """不驗簽，只看內容""" for name, part in zip(("header", "payload"), tok.split(".")[:2]): part += "=" * (-len(part) % 4) print(f"--- {name} ---") print(json.dumps(json.loads(base64.urlsafe_b64decode(part)), ensure_ascii=False, indent=2)) def call(url, token, method="GET", body=None, use_proxy=True): headers = {"Authorization": f"Bearer {token}", "Accept": "application/json"} data = None if body is not None: data = json.dumps(body).encode() headers["Content-Type"] = "application/json" req = urllib.request.Request(url, data=data, method=method, headers=headers) opener = (urllib.request.build_opener() if use_proxy else urllib.request.build_opener(urllib.request.ProxyHandler({}))) try: with opener.open(req, timeout=30) as r: print(f"{method} {r.status} 成功") print(r.read().decode("utf-8", "replace")[:1500]) except urllib.error.HTTPError as e: print(f"{method} 失敗 {e.code} {e.reason}") print("== 回應 headers（判斷是誰擋的關鍵）==") for k, v in e.headers.items(): print(f" {k}: {v}") print("== 回應 body ==") print(e.read().decode("utf-8", "replace")[:2000]) print("環境 proxy 設定:", urllib.request.getproxies()) token = get_token() # 沿用你原本的函式 decode_jwt(token) # ← 先看自己這顆 token 長什麼樣 url = f"{API_URL}?strSQLID={SQLID}" call(url, token) # GET call(url, token, use_proxy=False) # GET，繞過 proxy call(API_URL, token, "POST", {"strSQLID": SQLID}) # POST + JSON body
