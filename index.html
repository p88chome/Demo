POST 至少帶一個空 body，確保 method 判定為 POST method="POST", headers=headers ) try: with urllib.request.urlopen(req, timeout=30) as resp: print("status:", resp.status) data = json.loads(resp.read().decode("utf-8")) print(json.dumps(data, ensure_ascii=False, indent=2)[:1000]) return True except urllib.error.HTTPError as e: print("HTTP Error:", e.code, e.reason) print("Response headers:", dict(e.headers)) body_resp = e.read().decode("utf-8", errors="replace") print("Body:", repr(body_resp)) return False try: print("正在取得 token...") token = get_token() print("Token 取得成功:", token[:30], "...") base = f"{API_URL}?strSQLID={SQLID}" json_body = json.dumps({"strSQLID": SQLID}) #
