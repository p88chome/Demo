import urllib.request import urllib.parse import urllib.error import json TOKEN_URL = "http://f12aqiids.umc.com/connect/token" API_URL = "http://f12aqigw.umc.com/PI-Query/api/DataQueryAPI/GetQueryDataByJson" SQLID = "DQ-GFYPT-20260908-083610218884-MPZMJ" CLIENT_ID = "ADTUSER" CLIENT_SECRET = "ADTPROJECT01" SCOPE = "在這裡填入實際的 scope 值" # <-- 只要改這一行 def get_token(): payload = { "grant_type": "client_credentials", "client_id": CLIENT_ID, "client_secret": CLIENT_SECRET, "scope": SCOPE, } print("送出的 payload:", {**payload, "client_secret": "****"}) data = urllib.parse.urlencode(payload).encode("utf-8") req = urllib.request.Request( TOKEN_URL, data=data, method="POST", headers={"Content-Type": "application/x-www-form-urlencoded"} ) try: with urllib.request.urlopen(req, timeout=30) as resp: result = json.loads(resp.read().decode("utf-8")) print("換 token 成功") print("expires_in:", result.get("expires_in")) print("scope (伺服器回傳的):", result.get("scope")) return result.get("access_token") except urllib.error.HTTPError as e: print("換 token 失敗:", e.code, e.reason) print("Body:", e.read().decode("utf-8", errors="replace")) return None def get_data(token, sqlid): url = f"{API_URL}?strSQLID={sqlid}" req = urllib.request.Request( url, headers={ "Authorization": f"Bearer {token}", "Accept": "application/json", } ) try: with urllib.request.urlopen(req, timeout=30) as resp: print("查詢成功! status:", resp.status) data = json.loads(resp.read().decode("utf-8")) print(json.dumps(data, ensure_ascii=False, indent=2)[:2000]) return True except urllib.error.HTTPError as e: print("查詢失敗:", e.code, e.reason) print("Body:", e.read().decode("utf-8", errors="replace")) return False print("正在取得 token (含 scope)...") token = get_token() if token: print("-" * 60) print("正在查詢資料...") get_data(token, SQLID) else: print("因為換 token 失敗，不繼續查詢資料")
