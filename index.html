import urllib.request import urllib.parse import urllib.error import json TOKEN_URL = "http://f12aqiids.umc.com/connect/token" API_URL = "http://f12aqigw.umc.com/PI-Query/api/DataQueryAPI/GetQueryDataByJson" CLIENT_ID = "ADTUSER" CLIENT_SECRET = "ADTPROJECT01" SQLID_NO_BRACE = "DQ-GFYPT-20260908-083610218884-MPZMJ" SQLID_WITH_BRACE = "{DQ-GFYPT-20260908-083610218884-MPZMJ}" def get_token(): payload = { "grant_type": "client_credentials", "client_id": CLIENT_ID, "client_secret": CLIENT_SECRET, } data = urllib.parse.urlencode(payload).encode("utf-8") req = urllib.request.Request( TOKEN_URL, data=data, method="POST", headers={"Content-Type": "application/x-www-form-urlencoded"} ) with urllib.request.urlopen(req, timeout=30) as resp: result = json.loads(resp.read().decode("utf-8")) return result["access_token"] def try_call(label, sqlid, token): print("=" * 60) print(f"[{label}]") # 用 urlencode 確保大括號等特殊字元被正確編碼 encoded_sqlid = urllib.parse.quote(sqlid) url = f"{API_URL}?strSQLID={encoded_sqlid}" print("URL:", url) req = urllib.request.Request( url, headers={"Authorization": f"Bearer {token}", "Accept": "application/json"} ) try: with urllib.request.urlopen(req, timeout=30) as resp: print("狀態:", resp.status, "→ 成功!") data = json.loads(resp.read().decode("utf-8")) print(json.dumps(data, ensure_ascii=False, indent=2)[:1500]) return True except urllib.error.HTTPError as e: print("失敗:", e.code, e.reason) body = e.read().decode("utf-8", errors="replace") if body: print("Body:", body) return False print("正在取得 token...") token = get_token() print("Token 取得成功\n") try_call("測試1: 不帶大括號 (目前用法)", SQLID_NO_BRACE, token) try_call("測試2: 帶大括號 {}", SQLID_WITH_BRACE, token)
